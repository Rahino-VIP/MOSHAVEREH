لیست پیامک‌های ثبت‌شده در جدول sms_outbox

۱ کد ساخت و باز کردن دسترسی جدول `sms_outbox` در Supabase

اگر پیام‌ها در جدول ثبت نمی‌شوند، وارد پنل Supabase شوید، به SQL Editor بروید و کد زیر را Run کنید تا هم جدول و ستون‌ها کامل شوند و هم قفل RLS از روی جدول صف برداشته شود: 

CREATE TABLE IF NOT EXISTS public.sms_outbox (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  customer_name text,
  phone text NOT NULL,
  message text NOT NULL,
  reason_badge text,
  is_sent boolean DEFAULT false,
  created_at timestamp with time zone DEFAULT now()
);

ALTER TABLE public.sms_outbox ADD COLUMN IF NOT EXISTS customer_name text;
ALTER TABLE public.sms_outbox ADD COLUMN IF NOT EXISTS reason_badge text;

ALTER TABLE public.sms_outbox DISABLE ROW LEVEL SECURITY;
GRANT ALL ON public.sms_outbox TO anon, authenticated, service_role;
